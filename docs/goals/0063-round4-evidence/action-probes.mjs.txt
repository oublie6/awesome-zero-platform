import {test} from 'node:test';import assert from 'node:assert/strict';import {GameHandActions} from './native-input/clients/fair-doudizhu-cocos/dist/game/GameHandActions.js';import {CommandRejectedError} from './native-input/clients/fair-doudizhu-cocos/dist/game/GameClientShell.js';
const random={async bytes(n){return new Uint8Array(n).fill(11)}};
function fixture(callback=()=>{}){const shell={hand:{snapshot:{handId:'r4-controlled',phase:'PLAYING',version:10},canSubmit:true},async submitCommand(c){return {requestId:c.requestId}}};return {shell,actions:new GameHandActions(shell,random,callback)}}
test('observer exceptions cannot change committed-result handling or flags',async()=>{let calls=0;const f=fixture(()=>{calls++;throw Error('view disposed')});const r=await f.actions.play(['C3']);assert.ok(r.requestId);assert.equal(calls,2);assert.equal(f.actions.isSubmitting,false);assert.equal(f.actions.needsReconciliation,false)});
test('entropy cancellation clears isSubmitting without fabricating pending state',async()=>{const states=[];const f=fixture();const actions=new GameHandActions(f.shell,{async bytes(){throw Error('cancelled entropy')}},()=>states.push([actions.isSubmitting,actions.needsReconciliation]));await assert.rejects(actions.pass(),/cancelled/);assert.deepEqual(states,[[true,false],[false,false]])});
test('real rejection class clears pending; repeated unknown and exact request identity persist',async()=>{const seen=[],states=[];const f=fixture();const a=new GameHandActions(f.shell,random,()=>states.push([a.isSubmitting,a.needsReconciliation]));f.shell.submitCommand=async c=>{seen.push(structuredClone(c));throw Error('response lost')};await assert.rejects(a.pass(),/lost/);await assert.rejects(a.retryUnknown(),/lost/);assert.deepEqual(seen[0],seen[1]);assert.equal(a.needsReconciliation,true);assert.deepEqual(states,[[true,false],[false,true],[true,true],[false,true]]);f.shell.submitCommand=async c=>{seen.push(structuredClone(c));throw new CommandRejectedError('CONFLICT')};await assert.rejects(a.retryUnknown(),CommandRejectedError);assert.deepEqual(seen[0],seen[2]);assert.equal(a.needsReconciliation,false);assert.equal(a.isSubmitting,false);assert.deepEqual(states.at(-1),[false,false])});
test('discard requires newer authoritative snapshot and emits exactly one cleanup notification',async()=>{let calls=0;const f=fixture(()=>calls++);f.shell.submitCommand=async()=>{throw Error('lost')};await assert.rejects(f.actions.pass(),/lost/);assert.throws(()=>f.actions.discardUnknownAfterSync(),/not confirmed/);assert.equal(calls,2);f.shell.hand.snapshot={...f.shell.hand.snapshot,version:11};f.actions.discardUnknownAfterSync();assert.equal(calls,3);assert.equal(f.actions.needsReconciliation,false)});
